Cohen’s d

RPython
# R
cohens_d <- function(m1, m2, s1, s2, n1, n2) {
  pooled <- sqrt(((n1-1)*s1^2 + (n2-1)*s2^2) / (n1 + n2 - 2))
  (m2 - m1) / pooled
}
